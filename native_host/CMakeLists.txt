# Native host for the WebAssembly build: pd.wasm is translated to C with wasm2c and built together
# with the host code, so a native player runs the exact game the browsers run (and can join the
# same online matches). See README.md.
#
#   cmake -S native_host -B build-native -DWABT_DIR=/path/to/wabt [-DPD_WASM=build-web/pd.wasm]
#   cmake --build build-native -j
cmake_minimum_required(VERSION 3.16)
project(pdhost C)

set(PD_WASM "${CMAKE_CURRENT_SOURCE_DIR}/../build-web/pd.wasm" CACHE FILEPATH "the game's WebAssembly build (web/build.sh)")
set(WABT_DIR "" CACHE PATH "a built wabt checkout (wasm2c and its runtime)")
set(WASM2C "${WABT_DIR}/build/wasm2c" CACHE FILEPATH "the wasm2c tool")
set(PDHOST_PLATFORM "sdl" CACHE STRING "platform layer: sdl, or null (headless: no window, input or audio)")
set(PDHOST_PLATFORM_NAME "native" CACHE STRING "platform name the lobby sees")
set(PDHOST_DEFAULT_LOBBY "perfectdarklobby.m03.ca" CACHE STRING "lobby server of the Online menu")
option(PDHOST_TLS "secure connections (wss://) with mbedTLS" ON)

if(NOT EXISTS "${WABT_DIR}/wasm2c/wasm-rt-impl.c")
  message(FATAL_ERROR "set WABT_DIR to a wabt checkout with wasm2c built (https://github.com/WebAssembly/wabt)")
endif()

# pd.wasm -> C (8 files so the compiler can work in parallel)
set(W2C_DIR "${CMAKE_CURRENT_BINARY_DIR}/w2c")
set(W2C_OUT "${W2C_DIR}/pd.h")
foreach(i RANGE 7)
  list(APPEND W2C_OUT "${W2C_DIR}/pd_${i}.c")
endforeach()
add_custom_command(
  OUTPUT ${W2C_OUT}
  COMMAND ${CMAKE_COMMAND} -E make_directory "${W2C_DIR}"
  COMMAND "${WASM2C}" "${PD_WASM}" -o "${W2C_DIR}/pd.c" --module-name pd --num-outputs 8
  DEPENDS "${PD_WASM}"
  COMMENT "wasm2c: translating pd.wasm"
)
set_source_files_properties(${W2C_OUT} PROPERTIES GENERATED TRUE)

# pdbuild.h: the build id (clients and lobby must run the same pd.wasm) and the game state's
# memory ranges for match snapshots (pd.snap.json, written next to pd.wasm by the web build)
get_filename_component(PD_BUILD_DIR "${PD_WASM}" DIRECTORY)
set(PD_SNAP_JSON "${PD_BUILD_DIR}/pd.snap.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PD_WASM}" "${PD_SNAP_JSON}")
file(SHA1 "${PD_WASM}" PD_WASM_SHA1)
string(SUBSTRING "${PD_WASM_SHA1}" 0 12 PDHOST_BUILD_ID)
file(READ "${PD_SNAP_JSON}" PD_SNAP)
string(JSON PD_SNAP_COUNT LENGTH "${PD_SNAP}" ranges)
set(PDHOST_SNAP_RANGES "")
math(EXPR PD_SNAP_LAST "${PD_SNAP_COUNT} - 1")
foreach(i RANGE ${PD_SNAP_LAST})
  string(JSON start GET "${PD_SNAP}" ranges ${i} 0)
  string(JSON len GET "${PD_SNAP}" ranges ${i} 1)
  string(APPEND PDHOST_SNAP_RANGES "\t{ ${start}u, ${len}u },\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pdbuild.h"
  "// generated by native_host/CMakeLists.txt from pd.wasm and pd.snap.json\n"
  "#define PDHOST_BUILD_ID \"${PDHOST_BUILD_ID}\"\n"
  "#define PDHOST_DEFAULT_LOBBY \"${PDHOST_DEFAULT_LOBBY}\"\n"
  "#define PDHOST_PLATFORM_NAME \"${PDHOST_PLATFORM_NAME}\"\n"
  "#define PDHOST_SNAP_RANGE_COUNT ${PD_SNAP_COUNT}\n"
  "static const unsigned g_SnapRanges[][2] = {\n${PDHOST_SNAP_RANGES}};\n")
message(STATUS "pd.wasm build ${PDHOST_BUILD_ID}, ${PD_SNAP_COUNT} snapshot ranges")

add_executable(pdhost
  src/main.c
  src/fs.c
  src/gl.c
  src/net.c
  src/json.c
  src/ws.c
  src/platform_${PDHOST_PLATFORM}.c
  ${W2C_OUT}
  "${WABT_DIR}/wasm2c/wasm-rt-impl.c"
  "${WABT_DIR}/wasm2c/wasm-rt-mem-impl.c"
)
target_include_directories(pdhost PRIVATE "${W2C_DIR}" "${WABT_DIR}/wasm2c" src "${CMAKE_CURRENT_BINARY_DIR}")

# The game must compute exactly what the browsers compute: wasm arithmetic is IEEE-754 without
# fused multiply-add, so the C compiler must not contract or reassociate float math.
target_compile_options(pdhost PRIVATE -O2 -ffp-contract=off -fno-fast-math -fno-strict-aliasing
  $<$<C_COMPILER_ID:GNU>:-fexcess-precision=standard>)
set_source_files_properties(${W2C_OUT} PROPERTIES COMPILE_OPTIONS "-w")

if(PDHOST_PLATFORM STREQUAL "sdl")
  find_package(SDL2 REQUIRED)
  target_include_directories(pdhost PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(pdhost PRIVATE ${SDL2_LIBRARIES})
endif()
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
target_link_libraries(pdhost PRIVATE ZLIB::ZLIB Threads::Threads m)

if(PDHOST_TLS)
  include(FetchContent)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(mbedtls
    URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.4/mbedtls-3.6.4.tar.bz2)
  FetchContent_MakeAvailable(mbedtls)
  target_compile_definitions(pdhost PRIVATE PDHOST_TLS=1)
  target_link_libraries(pdhost PRIVATE mbedtls)
endif()
